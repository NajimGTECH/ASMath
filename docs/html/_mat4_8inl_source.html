433 Quaternion Mat4<T>::ExtractRotation() const
434 {
435 // Convert upper-left 3x3 to quaternion using standard matrix-to-quaternion conversion.
436 // Assumes matrix is rotation (possibly with scale). We remove scale first.
437 Vector3<T> scale = ExtractScale();
438 T sx = scale.x, sy = scale.y, sz = scale.z;
439 T invSx = (sx != static_cast<T>(0)) ? static_cast<T>(1) / sx : static_cast<T>(1);
440 T invSy = (sy != static_cast<T>(0)) ? static_cast<T>(1) / sy : static_cast<T>(1);
441 T invSz = (sz != static_cast<T>(0)) ? static_cast<T>(1) / sz : static_cast<T>(1);
442
443 T r00 = m[0][0] * invSx;
444 T r01 = m[0][1] * invSy;
445 T r02 = m[0][2] * invSz;
446 T r10 = m[1][0] * invSx;
447 T r11 = m[1][1] * invSy;
448 T r12 = m[1][2] * invSz;
449 T r20 = m[2][0] * invSx;
450 T r21 = m[2][1] * invSy;
451 T r22 = m[2][2] * invSz;
452
453 T trace = r00 + r11 + r22;
454 T qw, qx, qy, qz;
455
456 if (trace > static_cast<T>(0))
457 {
458 T s = std::sqrt(static_cast<double>(trace) + 1.0) * static_cast<T>(2);
459 qw = static_cast<T>(0.25) * s;
460 qx = (r21 - r12) / s;
461 qy = (r02 - r20) / s;
462 qz = (r10 - r01) / s;
463 }
464 else if ((r00 > r11) && (r00 > r22))
465 {
466 T s = std::sqrt(static_cast<double>(1.0 + r00 - r11 - r22)) * static_cast<T>(2);
467 qw = (r21 - r12) / s;
468 qx = static_cast<T>(0.25) * s;
469 qy = (r01 + r10) / s;
470 qz = (r02 + r20) / s;
471 }
472 else if (r11 > r22)
473 {
474 T s = std::sqrt(static_cast<double>(1.0 + r11 - r00 - r22)) * static_cast<T>(2);
475 qw = (r02 - r20) / s;
476 qx = (r01 + r10) / s;
477 qy = static_cast<T>(0.25) * s;

479 }
480 else

483 qw = (r10 - r01) / s;
484 qx = (r02 + r20) / s;
485 qy = (r12 + r21) / s;
486 qz = static_cast<T>(0.25) * s;
487 }
488

490 return Quaternion(static_cast<float>(qw), static_cast<float>(qx), static_cast<float>(qy), static_cast<float>(qz));

493 template <typename T>
494 bool Mat4<T>::ValidTRS() const
495 {
496 // Check last row is (0,0,0,1) and upper-left 3x3 is orthonormal up to scale.
497 const double EPS = 1e-5;
498 if (std::fabs(static_cast<double>(m[0][3])) > EPS) return false;
499 if (std::fabs(static_cast<double>(m[1][3])) > EPS) return false;
500 if (std::fabs(static_cast<double>(m[2][3])) > EPS) return false;
501 if (std::fabs(static_cast<double>(m[3][3] - static_cast<T>(1))) > EPS) return false;
502
503 Vector3<T> col0 = GetColumn(0);
504 Vector3<T> col1 = GetColumn(1);
505 Vector3<T> col2 = GetColumn(2);
506
507 // Non-zero scale?
508 Vector3<T> scale = ExtractScale();
509 if (scale.x == static_cast<T>(0) || scale.y == static_cast<T>(0) || scale.z == static_cast<T>(0))
510 return false;
511
512 // Normalize columns by scale and check orthonormality
513 Vector3<T> n0 = Vector3<T>(col0.x / scale.x, col0.y / scale.x, col0.z / scale.x).Normalized();
514 Vector3<T> n1 = Vector3<T>(col1.x / scale.y, col1.y / scale.y, col1.z / scale.y).Normalized();
515 Vector3<T> n2 = Vector3<T>(col2.x / scale.z, col2.y / scale.z, col2.z / scale.z).Normalized();
516
517 double d01 = std::fabs(static_cast<double>(n0.Dot(n1)));
518 double d02 = std::fabs(static_cast<double>(n0.Dot(n2)));
519 double d12 = std::fabs(static_cast<double>(n1.Dot(n2)));
520
521 if (d01 > EPS || d02 > EPS || d12 > EPS) return false;
522
523 return true;
524 }
